# CLI 工具实现计划

> **目标：** 为 Depsilo 新增完整的命令行接口，支持服务管理、状态查询、缓存操作和 shell 集成

**架构：** CLI 命令与 server 合并为同一个二进制。默认（无参数）启动服务器保持向后兼容，子命令执行 CLI 操作。CLI 命令通过 HTTP API 与运行中的 Depsilo 服务交互。

**设计原则：**
- 零新依赖 — 只用 Go 标准库 (`flag`, `net/http`, `os/exec`)
- 向后兼容 — `./bin/depsilo` (无参) 行为不变
- 所有 CLI 命令通过 HTTP 接口与服务的 admin API 交互

**涉及文件：**
- `cmd/depsilo/main.go` — 新建入口
- `internal/cli/` — CLI 命令包 (新建)
- `cmd/server/main_server.go` — 修改 (调 StartServer 逻辑)
- `cmd/server/server.go` — 不变 (StartServer 保持导出)
- `Makefile` — 修改 build 目标

func main() {
	// CLI mode if subcommand is provided and recognized
	if len(os.Args) > 1 {
		cmd := os.Args[1]
		switch cmd {
		case "serve", "server":
			// Fall through to server mode
		case "status", "activate", "start", "stop", "warmup", "flush", "version":
			os.Exit(cli.Run(cmd, os.Args[2:]))
		case "help", "--help", "-h":
			cli.PrintHelp()
			os.Exit(0)
		default:
			// Unknown subcommand — try as a non-CLI arg (could be pidfile for desktop, etc.)
			fmt.Fprintf(os.Stderr, "Unknown command: %s\n\n", cmd)
			cli.PrintHelp()
			os.Exit(1)
		}
	}

	// Server mode (default)
	logger, err := zap.NewProduction()
	if err != nil {
		fmt.Fprintf(os.Stderr, "failed to init logger: %v\n", err)
		os.Exit(1)
	}
	zap.ReplaceGlobals(logger)
	defer logger.Sync()

	ctx, cancel := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
	defer cancel()

	if err != nil {
		zap.L().Fatal("failed to start server", zap.Error(err))
	}

	<-ctx.Done()
	zap.L().Info("shutting down server...")
	if err := srv.Shutdown(context.Background()); err != nil {
		zap.L().Error("server shutdown error", zap.Error(err))
	}
}

    } `json:"service"`

        TotalRequests int64   `json:"total_requests"`
        HitCount      int64   `json:"hit_count"`
        MissCount     int64   `json:"miss_count"`
        HitRate       float64 `json:"hit_rate"`

    } `json:"today"`

        TotalSizeBytes  int64 `json:"total_size_bytes"`
    } `json:"cache"`
    Upstreams []struct {
        Name         string  `json:"name"`
        Adapter      string  `json:"adapter"`
        Healthy      bool    `json:"healthy"`

func formatBytes(b int64) string {
    switch {
    case b >= 1<<30:
        return fmt.Sprintf("%.1f GB", float64(b)/float64(1<<30))
    case b >= 1<<20:
        return fmt.Sprintf("%.0f MB", float64(b)/float64(1<<20))
    case b >= 1<<10:
        return fmt.Sprintf("%.0f KB", float64(b)/float64(1<<10))
    default:
        return fmt.Sprintf("%d B", b)
    }
}

func formatCount(n int64) string {
    if n < 1000 {
        return fmt.Sprintf("%d", n)
    }
    return fmt.Sprintf("%.1fk", float64(n)/1000)
}

### Task 5: 实现 `depsilo start` 和 `depsilo stop`

**Objective:** 管理 Depsilo 进程生命周期

**Files:**
- Create: `internal/cli/daemon.go`

**Step 1: 实现 `depsilo start [--daemon]`**

- 无 `--daemon`：== `depsilo serve`（前台运行，调 `server.StartServer`）
- 有 `--daemon`：后台启动，写 PID 文件到 `~/.local/share/depsilo/depsilo.pid`

```go
func runStart(args []string) int {
    daemon := false
    for _, a := range args {
        if a == "--daemon" || a == "-d" {
            daemon = true
        }
    }

    if !daemon {
        // Foreground: start server directly (same as `depsilo serve`)
        ctx, cancel := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer cancel()
        srv, err := server.StartServer(ctx)
        if err != nil {
            fmt.Fprintf(os.Stderr, "Error: %v\n", err)
            return 1
        }
        <-ctx.Done()
        srv.Shutdown(context.Background())
        return 0
    }

    // Daemon mode: fork and exec self
    pidFile, err := ensurePIDDir()
    if err != nil {
        fmt.Fprintf(os.Stderr, "Error: %v\n", err)
        return 1
    }

    // Check if already running
    if pid, _ := readPID(pidFile); pid > 0 {
        if processExists(pid) {
            fmt.Printf("Depsilo already running (PID %d)\n", pid)
            return 0
        }
        os.Remove(pidFile)
    }

    // Start child process
    cmd := exec.Command(os.Args[0], "serve")
    cmd.Stdout = nil  // detach stdout
    cmd.Stderr = nil  // detach stderr
    cmd.SysProcAttr = &syscall.SysProcAttr{Setpgid: true}

    if err := cmd.Start(); err != nil {
        fmt.Fprintf(os.Stderr, "Error starting daemon: %v\n", err)
        return 1
    }

    writePID(pidFile, cmd.Process.Pid)
    fmt.Printf("Depsilo started (PID %d)\n", cmd.Process.Pid)
    return 0
}
```

**Step 2: 实现 `depsilo stop`**

- 读取 PID 文件，发送 SIGTERM，等待退出

```go
func runStop(args []string) int {
    pidFile := getPIDFile()
    pid, err := readPID(pidFile)
    if err != nil {
        fmt.Fprintln(os.Stderr, "Depsilo is not running")
        return 1
    }

    process, err := os.FindProcess(pid)
    if err != nil {
        fmt.Fprintln(os.Stderr, "Depsilo is not running")
        os.Remove(pidFile)
        return 1
    }

    process.Signal(syscall.SIGTERM)
    // Wait for process to exit
    for i := 0; i < 10; i++ {
        if !processExists(pid) {
            break
        }
        time.Sleep(200 * time.Millisecond)
    }

    os.Remove(pidFile)
    fmt.Printf("Depsilo stopped (PID %d)\n", pid)
    return 0
}
```

**Step 3: 创建 pid 文件管理工具**

```go
func getPIDDir() string {
    // XDG规范: ~/.local/share/depsilo/
    home, _ := os.UserHomeDir()
    return filepath.Join(home, ".local", "share", "depsilo")
}

func getPIDFile() string {
    return filepath.Join(getPIDDir(), "depsilo.pid")
}
```

func runWarmup(args []string) int {
    if len(args) < 2 {
        fmt.Fprintln(os.Stderr, "Usage: depsilo warmup <ecosystem> <package> [package...]")
        fmt.Fprintln(os.Stderr, "Example: depsilo warmup pypi requests numpy torch")
        return 1
    }

    eco := args[0]
    pkgs := args[1:]

    body := map[string]any{
        "ecosystem": eco,
        "packages":  pkgs,
    }

    var result struct {
        Message  string `json:"message"`
        Packages int    `json:"packages"`
    }
    status, err := postJSON(getServerURL()+"/api/v1/admin/cache/warmup", body, &result)
    if err != nil {
        fmt.Fprintf(os.Stderr, "Error: %v\n", err)
        return 1
    }
    if status != 200 {
        fmt.Fprintf(os.Stderr, "Error: server returned %d\n", status)
        return 1
    }

    fmt.Printf("✓ Warmup started for %d packages (%s)\n", result.Packages, eco)
    fmt.Println("  Pre-fetching in background. Use 'depsilo status' to check cache.")
    return 0
}

### Task 8: 创建 cmd/depsilo/main.go 入口

**Objective:** 统一 CLI + Server 入口

**Files:**
- Create: `cmd/depsilo/main.go`

```go
//go:build !desktop

package main

import (
    "context"
    "fmt"
    "os"
    "os/signal"
    "syscall"

    "go.uber.org/zap"

    "depsilo/internal/cli"
    "depsilo/internal/server"
)

func main() {
    // CLI mode: if subcommand is recognized, dispatch
    if len(os.Args) > 1 {
        cmd := os.Args[1]
        switch cmd {
        case "serve", "server":
            // Fall through to server mode
        case "status", "activate", "start", "stop", "warmup", "flush", "version":
            os.Exit(cli.Run(cmd, os.Args[2:]))
        case "help", "--help", "-h":
            cli.PrintHelp()
            os.Exit(0)
        default:
            // Unknown subcommand — error
            fmt.Fprintf(os.Stderr, "Unknown command: %s\n\n", cmd)
            cli.PrintHelp()
            os.Exit(1)
        }
    }

    // Server mode (default)
    logger, err := zap.NewProduction()
    if err != nil {
        fmt.Fprintf(os.Stderr, "failed to init logger: %v\n", err)
        os.Exit(1)
    }
    zap.ReplaceGlobals(logger)
    defer logger.Sync()

    ctx, cancel := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
    defer cancel()

    srv, err := server.StartServer(ctx)
    if err != nil {
        zap.L().Fatal("failed to start server", zap.Error(err))
    }

    <-ctx.Done()
    zap.L().Info("shutting down server...")
    if err := srv.Shutdown(context.Background()); err != nil {
        zap.L().Error("server shutdown error", zap.Error(err))
    }
}
```
